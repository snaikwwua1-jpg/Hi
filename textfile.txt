                return "Sheriff"

                return "Murderer"

                    RemoveESP(Object)

        local function CreateESP(Object, Color, Core, Name)
            local Parts

            if typeof(Object) == "Instance" then
                Parts = Object:IsA("Model")
                    and Object:GetChildren()
                    or {Object}
            elseif typeof(Object) == "table" then
                Parts = Object
            else
                Parts = {}
            end

            local Boxes = {}
            local Billboard

            for _, Part in ipairs(Parts) do
                if Part:IsA("BasePart") then
                    local Box = Instance.new("BoxHandleAdornment")

                    Box.Size = Part.Size
                    Box.AlwaysOnTop = true
                    Box.ZIndex = 1
                    Box.AdornCullingMode = Enum.AdornCullingMode.Never
                    Box.Color3 = Color
                    Box.Transparency = 0.7
                    Box.Adornee = Part
                    Box.Parent = game.CoreGui

                    table.insert(Boxes, Box)

                    task.spawn(function()
                        while Box.Parent do
                            if not Box.Adornee
                                or not Box.Adornee:IsDescendantOf(Workspace) then
                                Box:Destroy()
                                break
                            end

                            task.wait(0.5)
                        end
                    end)
                end
            end

            if Core and Name then
                Billboard = Instance.new("BillboardGui")
                Billboard.AlwaysOnTop = true
                Billboard.Size = UDim2.new(0, 400, 0, 100)
                Billboard.Adornee = Core

                Label.Text = Name
                Label.Font = Enum.Font.GothamBold
                Label.TextSize = 14
                Label.Parent = Billboard

                Instance.new("UIStroke", Label)
            end

            return {
                Delete = function()
                    for _, Box in ipairs(Boxes) do
                        Box:Destroy()
                    end

                    if Billboard then
                        Billboard:Destroy()
                    end
                end
            }
        end

        local function RegisterESP(
            ToggleName,
            FlagKey,
            ListKey,
            ColorKey,
            CheckFunction,
            ParentTarget
        )
            DoorsTab:CreateToggle({
                Name = ToggleName,
                Default = false,
                Callback = function(Value)
                    Flags[FlagKey] = Value

                    if Value then
                        local Target = ParentTarget or Workspace.CurrentRooms

                        local function Scan(Object)
                            CheckFunction(Object, function(What, Core, Name)
                                table.insert(
                                    Lists[ListKey],
                                    CreateESP(What, Colors[ColorKey], Core, Name)
                                )
                            end)
                        end

                        local Connection = Target.DescendantAdded:Connect(Scan)

                        for _, Object in ipairs(Target:GetDescendants()) do
                            Scan(Object)
                        end

                        repeat
                            task.wait(0.5)
                        until not Flags[FlagKey]

                        Connection:Disconnect()
                        ClearList(Lists[ListKey])
                    else
                        ClearList(Lists[ListKey])
                    end
                end
            })
        end

                            Tracked[Object] = true

        GACFTab:CreateLabel({
            Text = "Note: If using Tower rebirth mode, keep your Rebirth UI open.",
            Color = Color3.fromRGB(200, 180, 100)
        })

        GACFTab:CreateDropdown({
            Name = "Rebirth Mode",
            Options = {
                "Normal Rebirth",
                "Tower Rebirth"
            },
            Default = "Normal Rebirth",
            Callback = function(Value)
                SelectedMode = Value
            end
        })

        GACFTab:CreateToggle({
            Name = "Auto Rebirth",
            Default = false,
            Callback = function(Value)
                AutoRebirth = Value

                if Value then
                    task.spawn(function()
                        while AutoRebirth do
                            if SelectedMode == "Normal Rebirth" then
                                if Remotes and Remotes:FindFirstChild("Rebirth") then
                                    pcall(function()
                                        Remotes.Rebirth:InvokeServer()
                                    end)
                                end

                                task.wait(0.3)
                            else
                                local Text = GetRebirthLabelText()

                                if Text:find("BACK TO YOUR COOP") then
                                    if Remotes
                                        and Remotes:FindFirstChild("TowerSurrender") then
                                        pcall(function()
                                            Remotes.TowerSurrender:InvokeServer()
                                        end)
                                    end

                                    repeat
                                        if Remotes
                                            and Remotes:FindFirstChild("Rebirth") then
                                            pcall(function()
                                                Remotes.Rebirth:InvokeServer()
                                            end)
                                        end

                                        task.wait(0.3)
                                    until Text:find("NOT YET")
                                        or not AutoRebirth
                                        or SelectedMode ~= "Tower Rebirth"
                                else
                                    task.wait(0.5)
                                end
                            end
                        end
                    end)
                end
            end
        })

        GACFTab:CreateToggle({
            Name = "Auto Buy Feeders",
            Default = false,
            Callback = function(Value)
                AutoBuyFeeders = Value

                if Value then
                    task.spawn(function()
                        while AutoBuyFeeders do
                            if Remotes
                                and Remotes:FindFirstChild("BuyGenerator") then

                                for Index = 1, 6 do
                                    if not AutoBuyFeeders then
                                        break
                                    end

                                    pcall(function()
                                        Remotes.BuyGenerator:InvokeServer(Index)
                                    end)

                                    task.wait(0.1)
                                end
                            end

                            task.wait(0.5)
                        end
                    end)
                end
            end
        })

        GACFTab:CreateToggle({
            Name = "Auto Upgrade Feeders",
            Default = false,
            Callback = function(Value)
                AutoUpgradeFeeders = Value

                if Value then
                    task.spawn(function()
                        while AutoUpgradeFeeders do
                            if Remotes
                                and Remotes:FindFirstChild("UpgradeGenerator") then

                                for Index = 1, 6 do
                                    if not AutoUpgradeFeeders then
                                        break
                                    end

                                    pcall(function()
                                        Remotes.UpgradeGenerator:InvokeServer(Index)
                                    end)

                                    task.wait(0.1)
                                end
                            end

                            task.wait(0.5)
                        end
                    end)
                end
            end
        })

        GACFTab:CreateToggle({
            Name = "Auto Expand Coop",
            Default = false,
            Callback = function(Value)
                AutoExpandCoop = Value

                if Value then
                    task.spawn(function()
                        while AutoExpandCoop do
                            if Remotes
                                and Remotes:FindFirstChild("ExpandCoop") then
                                pcall(function()
                                    Remotes.ExpandCoop:InvokeServer()
                                end)
                            end

                            task.wait(0.3)
                        end
                    end)
                end
            end
        })

        GACFTab:CreateToggle({
            Name = "Auto Upgrade Recycle",
            Default = false,
            Callback = function(Value)
                AutoUpgradeRecycle = Value

                if Value then
                    task.spawn(function()
                        while AutoUpgradeRecycle do
                            if Remotes
                                and Remotes:FindFirstChild("UpgradeRecycle") then
                                pcall(function()
                                    Remotes.UpgradeRecycle:InvokeServer()
                                end)
                            end

                            task.wait(0.3)
                        end
                    end)
                end
            end
        })

        GACFTab:CreateButton({
            Name = "Tower",
            Callback = function()
                if Remotes and Remotes:FindFirstChild("TowerStart") then
                    pcall(function()
                        Remotes.TowerStart:InvokeServer()
                    end)
                end
            end
        })

        GACFTab:CreateButton({
            Name = "Retreat Tower",
            Callback = function()
                if Remotes
                    and Remotes:FindFirstChild("TowerSurrender") then
                    pcall(function()
                        Remotes.TowerSurrender:InvokeServer()
                    end)
                end
            end
        })
    else
        GACFTab:CreateLabel({
            Text = "Join Grow a Chicken Fighter to unlock this tab",
            Color = Color3.fromRGB(200, 200, 200)
        })
    end
end)
